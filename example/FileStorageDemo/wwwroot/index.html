🔄 客户端直传流程
// 第一步：向后台请求签名 URL
GET /api/FileUploadWeb/GetClientUploadInfo?name=client-upload

// 后台返回签名上传信息
{
  "uploadUrl": "https://your-bucket.oss-cn-hangzhou.aliyuncs.com/client/xxx.png?OSSAccessKeyId=...&Expires=...&Signature=...",
  "method": "PUT",
  "fileFormName": "file",
  "headers": {}
}

// 第二步：客户端直接上传到存储服务
PUT {uploadUrl}
Content-Type: image/png

[文件二进制数据]

// 上传完成后文件直接在 OSS/MinIO 上，不经过后台！
